Analyzing NOAA Data Across New York State

data("ny_noaa")

ny_noaa_df = 
  ny_noaa |> 
  janitor::clean_names() |> 
  mutate(
    year = year(date),
    month = month(date, label = TRUE),
    prcp = prcp / 10,
    tmax = as.numeric(tmax) / 10,
    tmin = as.numeric(tmin) / 10) |>
  filter(year >= 2001)

set.seed(1)

ny_noaa_sample =
  ny_noaa_df |>
  drop_na(tmax, tmin) |>
  slice_sample(n = 3000)

Daily Maximum vs. Minimum Temperature, NY Weather Stations (2001–2010)

ny_noaa_sample |>
  plot_ly(
    x = ~tmin, y = ~tmax, type = "scatter", mode = "markers", color = ~month, alpha = 0.5) |>
  layout(
    title = "Daily Maximum vs. Minimum Temperature",
    xaxis = list(title = "Minimum Temperature (°C)"),
    yaxis = list(title = "Maximum Temperature (°C)")
  )

Daily Snowfall on Snowy Days by Month, NY (2001–2010)

set.seed(1)

ny_noaa_df |>
  filter(snow > 0) |>
  slice_sample(n = 3000) |>
  plot_ly(
    x = ~month, y = ~snow, color = ~month, type = "box", colors = "viridis") |>
  layout(
    title = "Daily Snowfall on Snowy Days by Month",
    xaxis = list(title = "Month"),
    yaxis = list(title = "Snowfall (mm, log scale)", type = "log"),
    showlegend = FALSE)
